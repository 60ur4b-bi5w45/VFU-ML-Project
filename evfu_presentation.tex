\section{Background \& Problem}

\begin{frame}{What is Vertical Federated Learning (VFL)?}
  \begin{itemize}
    \item In \textbf{horizontal} FL: each party has \emph{different samples} (rows) but same feature set.
    \item In \textbf{vertical} FL: each party has \emph{different features} (columns) for the \textbf{same entities} (e.g., same users, same patients).
  \end{itemize}
  \vspace{0.5em}
  \begin{block}{Example}
    Party A (hospital) has vitals and diagnoses; Party B (lab) has blood tests. Same patient ID, different columns. They train one model by sharing only \textbf{embeddings} or gradients, not raw data.
  \end{block}
  \vspace{0.5em}
  Used in healthcare, finance, and cross-institutional analytics where data cannot be centralized.
\end{frame}

\begin{frame}{Why Unlearning in VFL?}
  \begin{enumerate}
    \item \textbf{Right to erasure} (GDPR, etc.): A user can request that their data be removed not only from databases but also from any \emph{model trained on it}.
    \item In VFL, ``removing data'' can mean:
      \begin{itemize}
        \item removing a \textbf{client} (one party leaves),
        \item removing a \textbf{feature} (one column must no longer influence the model),
        \item or removing specific \textbf{samples} (rows).
      \end{itemize}
    \item Simply retraining from scratch is expensive; we need \textbf{unlearning} methods that update the model without full retraining.
  \end{enumerate}
  \vspace{0.5em}
  \textbf{Challenge:} How do we know unlearning actually worked? We need \textbf{verification}.
\end{frame}

\begin{frame}{Base Work: Varshney et al. (PETS 2025)}
  \begin{itemize}
    \item First \textbf{feature-level} unlearning methods for VFL: \textbf{VFU-KD} (knowledge distillation) and \textbf{VFU-GA} (gradient ascent).
    \item Idea: active party retrains (e.g., student mimics teacher on \emph{retain set only}, or gradient ascent on target samples); passive parties may freeze encoders.
    \item They evaluate unlearning using \textbf{output-logit membership inference}: can an attacker tell if a sample was in the training set by looking at model predictions?
  \end{itemize}
  \vspace{0.5em}
  \begin{alertblock}{Limitation}
    Output-only MIA is \textbf{weak}. Recent work (EMNLP 2025) shows that stronger, unlearning-specific MIA can reach $\sim$99\% accuracy even when standard metrics suggest ``successful'' unlearning. So we need \textbf{stronger verification}.
  \end{alertblock}
\end{frame}

\begin{frame}{Problem: Verification Gaps}
  \begin{enumerate}
    \item \textbf{Weak MIA}: Relying only on output logits misses leakage in gradients and internal representations.
    \item \textbf{No feature entanglement}: If features are correlated, removing one may still leave traces in other parties' encodings; no analysis in prior work.
    \item \textbf{No communication cost}: Overhead of unlearning in secure aggregation (SMPC/SecAgg) is unstudied.
    \item \textbf{Single-shot only}: No evaluation of repeated unlearning requests (iterative setting).
  \end{enumerate}
  \vspace{0.5em}
  \textbf{E-VFU} focuses on (1): we add \textbf{stronger adversarial verification} so we can rigorously check whether unlearning really removed the target data.
\end{frame}

\section{Our Approach: E-VFU}

\begin{frame}{E-VFU in One Slide}
  \begin{block}{What is E-VFU?}
    \textbf{Enhanced Vertical Federated Unlearning}: a \textbf{verification framework} that audits Varshney et al.'s VFU-KD/VFU-GA using four complementary checks. We do \emph{not} propose a new unlearning algorithm; we propose \emph{better ways to test} if unlearning worked.
  \end{block}
  \vspace{0.5em}
  \begin{block}{Phase 0 (base)}
    We use the same base as the paper: VFL model (bottom encoders per party + top fusion), training, and unlearning (e.g., gradient ascent on target samples). We also keep \textbf{output-based MIA} as a \textbf{baseline}.
  \end{block}
  \vspace{0.5em}
  \begin{block}{Phase 1 (robust verification)}
    We run \textbf{four} verification attacks. If all pass $\Rightarrow$ VFU unlearning is robust. If any fails $\Rightarrow$ we have evidence of remaining leakage.
  \end{block}
\end{frame}

\begin{frame}{Phase 1 — The Four Verification Components}
  \begin{enumerate}
    \item \textbf{Output MIA (baseline)}: Attacker uses prediction confidence/loss to guess membership. \emph{Target:} AUC $\approx$ 0.5 on unlearned samples (cannot distinguish from non-members).
    \item \textbf{Gradient-based MIA}: Attacker uses \emph{gradient} norms and spectral features to guess membership. Stronger than output-only. \emph{Target:} AUC $\approx$ 0.5; we also report gradient change $\|g_{\text{before}}\| - \|g_{\text{after}}\|$.
    \item \textbf{FUMA}: Forensic Unlearning Membership Attack — use \emph{canary} samples and prediction divergence $\Delta_{\text{pred}}$ to detect unlearning artifacts. \emph{Target:} detection rate $<$ 5\%.
    \item \textbf{Shadow model attack}: Train shadow models on retain-only vs full data; meta-classifier predicts which dataset a model came from. \emph{Target:} unlearned model classified as retain-only (P(full data) $<$ 0.5).
  \end{enumerate}
\end{frame}

\begin{frame}{Why These Four?}
  \begin{itemize}
    \item \textbf{Output MIA}: Standard baseline (Varshney et al.); easy to compare.
    \item \textbf{Gradient MIA}: Gradients reveal more than logits (Geiping et al.); if unlearning is real, gradient response to unlearned samples should change.
    \item \textbf{FUMA}: Unlearning can leave \emph{behavioral} traces (e.g., sensitivity to canaries); FUMA checks that such traces are gone.
    \item \textbf{Shadow attack}: Checks the \emph{whole model}, not per-sample membership — is the unlearned model statistically indistinguishable from a model that never saw the target data?
  \end{itemize}
  \vspace{0.5em}
  Together they cover: \emph{outputs}, \emph{gradients}, \emph{behavior on probes}, and \emph{model-level fingerprint}.
\end{frame}

\begin{frame}{Phase 1 — Success Criteria (Summary Table)}
  \begin{center}
    \small
    \begin{tabular}{@{}p{3.2cm}p{3.2cm}p{3.8cm}@{}}
      \toprule

\end{frame}

\section{Experimental Setup}

\begin{frame}{Experimental Setup}
  \begin{itemize}
    \item \textbf{Datasets}: UCI Adult (income prediction, $\sim$32K samples), German Credit (credit risk, 1K); optional MIMIC-III (healthcare, 46K).
    \item \textbf{VFL}: Data split by \textbf{features} across 2 (or 3) parties; one active party holds labels.
    \item \textbf{Model}: Bottom MLPs per party $\to$ embeddings $\to$ top fusion MLP $\to$ binary prediction.
    \item \textbf{Unlearning}: We use \textbf{VFU-GA} (gradient ascent on target samples for a fixed number of epochs, e.g., 10). A fraction (e.g., 5\%) of training samples are selected as ``target'' to unlearn.
    \item \textbf{Verification}: After unlearning, we run all four Phase 1 attacks and compare metrics to the criteria above.
  \end{itemize}
\end{frame}

\section{Conclusions}

\begin{frame}{What We Did}
  \begin{itemize}
    \item We implemented \textbf{Phase 0} (VFL model, training, VFU-GA unlearning) and \textbf{Phase 1} (four verification attacks) as in the E-VFU proposal.
    \item We compared \textbf{output-based MIA} (baseline) with \textbf{gradient-based MIA}, and added \textbf{FUMA} and \textbf{shadow model} verification.
    \item Each attack has a clear \textbf{success criterion}; we report whether unlearning passes or fails each check.
    \item All plots are saved separately in \texttt{presentation\_figures/} (numbered 01--08) for easy inclusion in reports and slides.
  \end{itemize}
\end{frame}

\begin{frame}{Interpretation \& Takeaway}
  \begin{block}{If verification passes (all four checks)}
    E-VFU provides evidence that Varshney et al.'s VFU unlearning \textbf{removes} the target data's influence from the model, not only in terms of output logits but also in gradients, forensic behavior, and model-level statistics. The baseline (output MIA) alone would be insufficient to conclude this.
  \end{block}
  \vspace{0.5em}
  \begin{block}{If verification fails (one or more checks)}
    E-VFU \textbf{reveals limitations}: some trace of the target data remains (e.g., in gradients or in the shadow classifier's view). That motivates stronger unlearning methods or more epochs, and shows the value of multi-attack verification.
  \end{block}
  \vspace{0.5em}
  \textbf{Bottom line:} Stronger verification leads to more reliable conclusions about whether unlearning actually worked.
\end{frame}
